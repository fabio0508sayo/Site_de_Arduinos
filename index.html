MOD 01 // ILUMINAÇÃO INTELIGENTE

PIR LDR LED

Sistema automatizado que aciona a iluminação apenas quando detecta movimento (PIR) em ambientes com baixa luminosidade (LDR). Economia e inteligência.

FIRMWARE


const int PIR_PIN = 2;
const int LDR_PIN = A0;
const int LED_PIN = 13;

void setup() {
  pinMode(PIR_PIN, INPUT);
  pinMode(LED_PIN, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  int ldrValue = analogRead(LDR_PIN);
  int motion = digitalRead(PIR_PIN);
  
  if (ldrValue < 500 && motion == HIGH) {
    digitalWrite(LED_PIN, HIGH);
  } else {
    digitalWrite(LED_PIN, LOW);
  }
  delay(100);
}

MOD 02 // SENSOR DE DISTÂNCIA

ULTRASSÔNICO POTENCIÔMETRO LED

Mede a distância de objetos próximos. O potenciômetro ajusta a sensibilidade de disparo do LED, indicando proximidade de perigo.

FIRMWARE


#include <NewPing.h>

#define TRIGGER_PIN  3
#define ECHO_PIN     4
#define MAX_DISTANCE 200

NewPing sonar(TRIGGER_PIN, ECHO_PIN, MAX_DISTANCE);
const int POT_PIN = A1;
const int LED_PIN = 12;

void setup() {
  pinMode(LED_PIN, OUTPUT);
}

void loop() {
  int threshold = map(analogRead(POT_PIN), 0, 1023, 5, 100);
  int distance = sonar.ping_cm();
  
  if (distance > 0 && distance < threshold) {
    digitalWrite(LED_PIN, HIGH);
  } else {
    digitalWrite(LED_PIN, LOW);
  }
  delay(50);
}

MOD 03 // CLIMA CONTROL

TEMPERATURA UMIDADE LEDS

Monitoramento contínuo das condições ambientais. LEDs de diferentes cores alertam se a temperatura ou umidade ultrapassarem os limites ideais.

FIRMWARE


#include "DHT.h"

#define DHTPIN 5
#define DHTTYPE DHT11

DHT dht(DHTPIN, DHTTYPE);
const int LED_TEMP_ALARM = 11;
const int LED_HUM_ALARM = 10;

void setup() {
  dht.begin();
  pinMode(LED_TEMP_ALARM, OUTPUT);
  pinMode(LED_HUM_ALARM, OUTPUT);
}

void loop() {
  float h = dht.readHumidity();
  float t = dht.readTemperature();
  
  if (isnan(h) || isnan(t)) return;
  
  digitalWrite(LED_TEMP_ALARM, t > 30.0 ? HIGH : LOW);
  digitalWrite(LED_HUM_ALARM, h < 30.0 || h > 70.0 ? HIGH : LOW);
  
  delay(2000);
}
